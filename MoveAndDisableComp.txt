Import-Module ActiveDirectory

# Configuration
$CsvPath = "C:\Users\Administrator\Documents\ComputersList.txt"
$TargetOU = "OU=Computers,OU=Disabled Objects,DC=local,DC=int"
$WhatIfMode = $true  # $true = preview, $false = execute

# Report location
$ReportFolder = "C:\Users\Administrator\Documents\ADCleanupReports"

# Use one writable domain controller
$ADServer = (Get-ADDomainController -Discover -Writable -ErrorAction Stop).HostName

# Validate OU and import CSV
Get-ADOrganizationalUnit -Identity $TargetOU -Server $ADServer -ErrorAction Stop | Out-Null
$Computers = Import-Csv -Path $CsvPath -ErrorAction Stop

# Create report folder
New-Item -Path $ReportFolder -ItemType Directory -Force | Out-Null

# Process computers
$Results = foreach ($Computer in $Computers) {

    $Name = if ($null -ne $Computer.ComputerName) {
        $Computer.ComputerName.Trim()
    } else {
        ""
    }

    if ([string]::IsNullOrWhiteSpace($Name)) {
        [PSCustomObject]@{
            ComputerName = "(blank)"
            Status       = "SKIPPED"
            Details      = "Empty computer name"
        }
        continue
    }

    try {
        # Find computer account
        $ADComputer = Get-ADComputer -Identity $Name `
            -Properties Enabled, DistinguishedName `
            -Server $ADServer -ErrorAction Stop

        # Check current status
        $Disabled = -not $ADComputer.Enabled

        $InTargetOU = $ADComputer.DistinguishedName.EndsWith(
            ",$TargetOU",
            [StringComparison]::OrdinalIgnoreCase
        )

        # Skip if already complete
        if ($Disabled -and $InTargetOU) {
            $Status = "ALREADY DONE"
            $Details = "Disabled and already in target OU"
        }
        else {
            $Actions = @()

            # Disable account if necessary
            if (-not $Disabled) {
                $CurrentAction = "Disable account"

                Disable-ADAccount -Identity $ADComputer.ObjectGUID `
                    -Server $ADServer `
                    -WhatIf:$WhatIfMode `
                    -ErrorAction Stop

                $Actions += "Disable"
            }

            # Move object if necessary
            if (-not $InTargetOU) {
                $CurrentAction = "Move computer object"

                Move-ADObject -Identity $ADComputer.ObjectGUID `
                    -TargetPath $TargetOU `
                    -Server $ADServer `
                    -WhatIf:$WhatIfMode `
                    -ErrorAction Stop

                $Actions += "Move"
            }

            if ($WhatIfMode) {
                $Status = "PREVIEW"
                $Details = "Would perform: $($Actions -join ' and ')"
            }
            else {
                # Verify final AD state
                $CurrentAction = "Verify changes"

                $UpdatedComputer = Get-ADComputer -Identity $ADComputer.ObjectGUID `
                    -Properties Enabled, DistinguishedName `
                    -Server $ADServer -ErrorAction Stop

                $VerifiedDisabled = -not $UpdatedComputer.Enabled

                $VerifiedInOU = $UpdatedComputer.DistinguishedName.EndsWith(
                    ",$TargetOU",
                    [StringComparison]::OrdinalIgnoreCase
                )

                if ($VerifiedDisabled -and $VerifiedInOU) {
                    $Status = "SUCCESS"
                    $Details = "Computer disabled and moved successfully; changes verified"
                }
                else {
                    $Status = "VERIFY FAILED"
                    $Details = "Final state does not match requirements. Disabled: $VerifiedDisabled; In target OU: $VerifiedInOU"
                }
            }
        }

        [PSCustomObject]@{
            ComputerName = $Name
            Status       = $Status
            Details      = $Details
        }
    }
    catch {
        [PSCustomObject]@{
            ComputerName = $Name
            Status       = "FAILED"
            Details      = "$CurrentAction failed: $($_.Exception.Message)"
        }
    }
}

# Generate timestamped report
$Timestamp = Get-Date -Format "yyyyMMdd"
$ReportPath = "$ReportFolder\AD_MoveDisable_Report_$Timestamp.csv"

# Export all results
$Results | Export-Csv -Path $ReportPath -NoTypeInformation

# Display summary
Write-Host "`n========================================" -ForegroundColor DarkGray
Write-Host " ACTIVE DIRECTORY DISABLE / MOVE SUMMARY" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor DarkGray

$Results | Group-Object Status | Sort-Object Name |
    ForEach-Object {
        Write-Host ("{0,-15} {1}" -f $_.Name, $_.Count)
    }

Write-Host "`nTotal processed: $($Results.Count)"

# Display exceptions
$Exceptions = @($Results | Where-Object {
    $_.Status -in @("FAILED", "VERIFY FAILED", "SKIPPED")
})

if ($Exceptions.Count -gt 0) {
    Write-Host "`n========== EXCEPTIONS ==========" -ForegroundColor Yellow

    $Exceptions |
        Format-Table ComputerName, Status, Details -AutoSize -Wrap
}
else {
    Write-Host "`nNo exceptions encountered." -ForegroundColor Green
}

# Report location
Write-Host "`nFull report saved to:" -ForegroundColor Cyan
Write-Host $ReportPath -ForegroundColor Green

Write-Host "`n========================================" -ForegroundColor DarkGray
