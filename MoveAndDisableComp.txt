
Import-Module ActiveDirectory

# Configuration
$CsvPath = "C:\Users\Administrator\Documents\ComputersList.txt"
$TargetOU = "OU=Computers,OU=Disabled Objects,DC=local,DC=int"
$WhatIfMode = $true  # $true = preview, $false = execute

# Validate OU and import CSV
Get-ADOrganizationalUnit -Identity $TargetOU -ErrorAction Stop | Out-Null
$Computers = Import-Csv -Path $CsvPath -ErrorAction Stop

# Process computers
$Results = foreach ($Computer in $Computers) {
    $Name = $Computer.ComputerName.Trim()

    if ([string]::IsNullOrWhiteSpace($Name)) {
        [PSCustomObject]@{
            ComputerName = "(blank)"
            Status       = "SKIPPED"
            Details      = "Empty computer name"
        }
        continue
    }

    try {
        $ADComputer = Get-ADComputer -Identity $Name `
            -Properties Enabled, DistinguishedName -ErrorAction Stop

        $Disabled = -not $ADComputer.Enabled
        $InTargetOU = $ADComputer.DistinguishedName.EndsWith(
            ",$TargetOU",
            [StringComparison]::OrdinalIgnoreCase
        )

        if ($Disabled -and $InTargetOU) {
            $Status = "ALREADY DONE"
            $Details = "Disabled and already in target OU"
        }
        else {
            if (-not $Disabled) {
                Disable-ADAccount -Identity $ADComputer `
                    -WhatIf:$WhatIfMode -ErrorAction Stop
            }

            if (-not $InTargetOU) {
                Move-ADObject -Identity $ADComputer `
                    -TargetPath $TargetOU `
                    -WhatIf:$WhatIfMode -ErrorAction Stop
            }

            $Status = if ($WhatIfMode) { "PREVIEW" } else { "SUCCESS" }
            $Details = if ($WhatIfMode) {
                "Would disable and/or move as needed"
            } else {
                "Disabled and/or moved as needed"
            }
        }

        [PSCustomObject]@{
            ComputerName = $Name
            Status       = $Status
            Details      = $Details
        }
    }
    catch {
        [PSCustomObject]@{
            ComputerName = $Name
            Status       = "FAILED"
            Details      = $_.Exception.Message
        }
    }
}

# Display results and summary
$Results | Format-Table ComputerName, Status, Details -AutoSize -Wrap

Write-Host "`nSUMMARY" -ForegroundColor Cyan
$Results | Group-Object Status | Sort-Object Name |
    ForEach-Object {
        Write-Host ("{0,-15} {1}" -f $_.Name, $_.Count)
    }
