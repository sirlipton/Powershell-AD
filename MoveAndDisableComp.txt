Import-Module ActiveDirectory -ErrorAction Stop

# ============================================================
# CONFIGURATION
# ============================================================

$CsvPath = "C:\Users\Administrator\Documents\PowerShell\CompBatch1.txt"

$TargetOU = "OU=Computers,OU=Disabled Objects,DC=local,DC=int"

$ADServer = "local.int"

$WhatIfMode = $true  # $true = preview, $false = execute

$ReportFolder = "C:\Users\Administrator\Documents\ADCleanupReports"

# ============================================================
# INITIAL SETUP
# ============================================================

$Results = [System.Collections.Generic.List[object]]::new()

# Track computer names to identify duplicates.
# Duplicates will still be processed.
$SeenComputers = @{}

$Timestamp = Get-Date -Format "yyyyMMdd_HHmmss"

$ReportPath = Join-Path $ReportFolder "AD_MoveDisable_Report_$Timestamp.csv"

# Create report folder if it does not exist.
if (-not (Test-Path $ReportFolder)) {
    New-Item -Path $ReportFolder -ItemType Directory -Force | Out-Null
}

Write-Host ""
Write-Host "==================================================" -ForegroundColor Cyan
Write-Host " ACTIVE DIRECTORY COMPUTER DISABLE / MOVE REPORT" -ForegroundColor Cyan
Write-Host "==================================================" -ForegroundColor Cyan

Write-Host "AD Server: $ADServer"
Write-Host "Destination: $TargetOU"

if ($WhatIfMode) {
    Write-Host "Mode: PREVIEW ONLY" -ForegroundColor Yellow
}
else {
    Write-Host "Mode: LIVE EXECUTION" -ForegroundColor Red
}

Write-Host ""

# ============================================================
# VALIDATE CONFIGURATION AND INPUT
# ============================================================

try {
    Get-ADOrganizationalUnit -Identity $TargetOU -Server $ADServer -ErrorAction Stop | Out-Null

    $ComputerList = Import-Csv -Path $CsvPath -ErrorAction Stop

    if (-not $ComputerList) {
        throw "The CSV file is empty."
    }

    if (-not ($ComputerList[0].PSObject.Properties.Name -contains "ComputerName")) {
        throw "CSV file must contain a column named ComputerName."
    }
}
catch {
    Write-Host "SETUP FAILED: $($_.Exception.Message)" -ForegroundColor Red
    return
}

# ============================================================
# PROCESS COMPUTERS
# ============================================================

foreach ($Computer in $ComputerList) {

    $Name = ([string]$Computer.ComputerName).Trim()

    if ([string]::IsNullOrWhiteSpace($Name)) {

        $Results.Add([PSCustomObject]@{
            ComputerName = "(Blank)"
            Status       = "SKIPPED"
            Duplicate    = "NO"
            Details      = "Computer name is blank"
        })

        continue
    }

    # Check whether this computer name appeared previously.
    # Do not skip it if it is a duplicate.
    $DuplicateKey = $Name.ToUpperInvariant()
    $IsDuplicate = $SeenComputers.ContainsKey($DuplicateKey)

    if ($IsDuplicate) {
        Write-Host "DUPLICATE ENTRY: $Name - Processing anyway" -ForegroundColor Yellow
    }
    else {
        $SeenComputers[$DuplicateKey] = $true
    }

    $DuplicateStatus = if ($IsDuplicate) { "YES" } else { "NO" }

    $CurrentAction = "Read computer from Active Directory"

    try {

        Write-Host "Processing: $Name" -ForegroundColor Cyan

        # Retrieve computer object.
        $ADComputer = Get-ADComputer -Identity $Name `
            -Properties Enabled, DistinguishedName `
            -Server $ADServer `
            -ErrorAction Stop

        $IsDisabled = -not $ADComputer.Enabled

        $IsInTargetOU = $ADComputer.DistinguishedName.EndsWith(
            ",$TargetOU",
            [System.StringComparison]::OrdinalIgnoreCase
        )

        # Determine required actions.
        $Actions = @()

        if ($IsDisabled -and $IsInTargetOU) {

            $Results.Add([PSCustomObject]@{
                ComputerName = $Name
                Status       = "ALREADY DONE"
                Duplicate    = $DuplicateStatus
                Details      = "Computer is already disabled and in the destination OU"
            })

            Write-Host "Already disabled and in destination OU." -ForegroundColor Green
            continue
        }

        if (-not $IsDisabled) {
            $Actions += "Disable"
        }

        if (-not $IsInTargetOU) {
            $Actions += "Move"
        }

        # Disable computer if necessary.
        if (-not $IsDisabled) {

            $CurrentAction = "Disable computer"

            Disable-ADAccount -Identity $ADComputer.ObjectGUID `
                -Server $ADServer `
                -WhatIf:$WhatIfMode `
                -ErrorAction Stop
        }

        # Move computer if necessary.
        if (-not $IsInTargetOU) {

            $CurrentAction = "Move computer"

            Move-ADObject -Identity $ADComputer.ObjectGUID `
                -TargetPath $TargetOU `
                -Server $ADServer `
                -WhatIf:$WhatIfMode `
                -ErrorAction Stop
        }

        # Preview result.
        if ($WhatIfMode) {

            $Results.Add([PSCustomObject]@{
                ComputerName = $Name
                Status       = "PREVIEW"
                Duplicate    = $DuplicateStatus
                Details      = "Would perform: $($Actions -join ', ')"
            })

            Write-Host "Preview completed: $($Actions -join ', ')" -ForegroundColor Yellow
        }

        # Verify actual changes.
        else {

            $CurrentAction = "Verify computer status"

            $VerifiedComputer = Get-ADComputer -Identity $ADComputer.ObjectGUID `
                -Properties Enabled, DistinguishedName `
                -Server $ADServer `
                -ErrorAction Stop

            $VerifiedDisabled = -not $VerifiedComputer.Enabled

            $VerifiedInTargetOU = $VerifiedComputer.DistinguishedName.EndsWith(
                ",$TargetOU",
                [System.StringComparison]::OrdinalIgnoreCase
            )

            if ($VerifiedDisabled -and $VerifiedInTargetOU) {

                $Results.Add([PSCustomObject]@{
                    ComputerName = $Name
                    Status       = "SUCCESS"
                    Duplicate    = $DuplicateStatus
                    Details      = "Successfully disabled and moved to destination OU"
                })

                Write-Host "SUCCESS: $Name" -ForegroundColor Green
            }
            else {

                $VerificationDetails = "Disabled: $VerifiedDisabled; In Target OU: $VerifiedInTargetOU"

                $Results.Add([PSCustomObject]@{
                    ComputerName = $Name
                    Status       = "VERIFY FAILED"
                    Duplicate    = $DuplicateStatus
                    Details      = $VerificationDetails
                })

                Write-Host "VERIFY FAILED: $Name" -ForegroundColor Red
            }
        }
    }
    catch {

        $Results.Add([PSCustomObject]@{
            ComputerName = $Name
            Status       = "FAILED"
            Duplicate    = $DuplicateStatus
            Details      = "$CurrentAction failed: $($_.Exception.Message)"
        })

        Write-Host "FAILED: $Name - $($_.Exception.Message)" -ForegroundColor Red
    }
}

# ============================================================
# EXPORT FULL REPORT
# ============================================================

$Results | Export-Csv -Path $ReportPath -NoTypeInformation -Encoding UTF8

# ============================================================
# SUMMARY
# ============================================================

Write-Host ""
Write-Host "==================================================" -ForegroundColor Cyan
Write-Host " SUMMARY" -ForegroundColor Cyan
Write-Host "==================================================" -ForegroundColor Cyan

$Results |
    Group-Object Status |
    Sort-Object Name |
    ForEach-Object {
        Write-Host ("{0,-18} {1}" -f $_.Name, $_.Count)
    }

$DuplicateCount = @($Results | Where-Object {
    $_.Duplicate -eq "YES"
}).Count

Write-Host ""
Write-Host "Duplicate entries: $DuplicateCount" -ForegroundColor Yellow

# ============================================================
# EXCEPTIONS REPORT
# ============================================================

$Exceptions = @($Results | Where-Object {

    $_.Status -in @("FAILED", "VERIFY FAILED", "SKIPPED") -or
    $_.Duplicate -eq "YES"

})

Write-Host ""
Write-Host "==================================================" -ForegroundColor Cyan
Write-Host " EXCEPTIONS" -ForegroundColor Cyan
Write-Host "==================================================" -ForegroundColor Cyan

if ($Exceptions.Count -gt 0) {

    $Exceptions |
        Format-Table ComputerName, Status, Duplicate, Details -AutoSize -Wrap

}
else {

    Write-Host "No exceptions found." -ForegroundColor Green
}

Write-Host ""
Write-Host "Full report saved to:"
Write-Host $ReportPath -ForegroundColor Green

Write-Host ""
Write-Host "Processing complete." -ForegroundColor Cyan
