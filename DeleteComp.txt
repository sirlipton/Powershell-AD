Import-Module ActiveDirectory -ErrorAction Stop

# Configuration
$CsvPath = "C:\Users\Administrator\Documents\PowerShell\CompBatch1.txt" #Change filepath as needed
$DisabledOU = "OU=Computers,OU=Disabled Objects,DC=local,DC=int"
$ADServer = "local.int"
$WhatIfMode = $true  #true = test, false = delete
$ReportFolder = "C:\Users\Administrator\Documents\ADCleanupReports"

# Setup
$Results = [System.Collections.Generic.List[object]]::new()
$SeenComputers = @{}
$Timestamp = Get-Date -Format "yyyyMMdd"
$ReportPath = Join-Path $ReportFolder "AD_Comp_Deletion_Report_$Timestamp.csv"

function Add-Result {
    param($Name, $Status, $Duplicate, $Details)

    $Results.Add([PSCustomObject]@{
        ComputerName = $Name
        Status       = $Status
        Duplicate    = $Duplicate
        Details      = $Details
    })
}

# Validate OU and CSV
try {
    Get-ADOrganizationalUnit -Identity $DisabledOU -Server $ADServer -ErrorAction Stop | Out-Null
    $Computers = Import-Csv -Path $CsvPath -ErrorAction Stop

    if (-not $Computers) { throw "CSV file is empty." }
    if ($Computers[0].PSObject.Properties.Name -notcontains "ComputerName") {
        throw "CSV must contain a ComputerName column."
    }
}
catch {
    Write-Host "SETUP FAILED: $($_.Exception.Message)" -ForegroundColor Red
    return
}

New-Item -Path $ReportFolder -ItemType Directory -Force | Out-Null

Write-Host "`nAD Computer Deletion - $ADServer" -ForegroundColor Cyan
Write-Host "Required OU: $DisabledOU"

if ($WhatIfMode) {
    Write-Host "Mode: PREVIEW ONLY" -ForegroundColor Yellow
}
else {
    Write-Host "Mode: LIVE DELETION" -ForegroundColor Red
}

# Process computers
foreach ($Computer in $Computers) {

    $Name = ([string]$Computer.ComputerName).Trim()

    if ([string]::IsNullOrWhiteSpace($Name)) {
        Add-Result "(Blank)" "SKIPPED" "NO" "Empty computer name"
        continue
    }

    # Detect duplicates but still process them
    $Key = $Name.ToUpperInvariant()
    $IsDuplicate = $SeenComputers.ContainsKey($Key)
    $SeenComputers[$Key] = $true
    $Duplicate = if ($IsDuplicate) { "YES" } else { "NO" }

    try {
        Write-Host "`nProcessing: $Name" -ForegroundColor Cyan

        $ADComputer = Get-ADComputer -Identity $Name `
            -Properties DistinguishedName -Server $ADServer -ErrorAction Stop

        # Only delete computers in the Disabled OU or its child OUs
        if (-not $ADComputer.DistinguishedName.EndsWith(
            ",$DisabledOU",
            [System.StringComparison]::OrdinalIgnoreCase
        )) {
            Add-Result $Name "BLOCKED" $Duplicate "Outside required Disabled OU"
            continue
        }

        Remove-ADComputer -Identity $ADComputer -Server $ADServer `
            -WhatIf:$WhatIfMode -Confirm:$false -ErrorAction Stop

        if ($WhatIfMode) {
            Add-Result $Name "PREVIEW" $Duplicate "Would delete"
        }
        else {
            Add-Result $Name "DELETED" $Duplicate "Deletion command completed"
        }
    }
    catch {
        Add-Result $Name "FAILED" $Duplicate $_.Exception.Message
    }
}

# Export report
$Results | Export-Csv -Path $ReportPath -NoTypeInformation -Encoding UTF8

# Summary
Write-Host "`n========== SUMMARY ==========" -ForegroundColor Cyan

$Results | Group-Object Status | Sort-Object Name |
    ForEach-Object {
        Write-Host ("{0,-15} {1}" -f $_.Name, $_.Count)
    }

$DuplicateCount = @($Results | Where-Object Duplicate -eq "YES").Count
Write-Host "`nTotal processed: $($Results.Count)"
Write-Host "Duplicate entries: $DuplicateCount" -ForegroundColor Yellow

# Exceptions
$Exceptions = @($Results | Where-Object {
    $_.Status -in @("FAILED", "BLOCKED", "SKIPPED") -or $_.Duplicate -eq "YES"
})

Write-Host "`n========== EXCEPTIONS ==========" -ForegroundColor Yellow

if ($Exceptions.Count) {
    $Exceptions | Format-Table ComputerName, Status, Duplicate, Details -AutoSize -Wrap
}
else {
    Write-Host "No exceptions found." -ForegroundColor Green
}

Write-Host "`nReport saved to: $ReportPath" -ForegroundColor Green
